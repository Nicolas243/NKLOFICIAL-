```typescript
  search: string;
```

```typescript
  const [isFavorited, setIsFavorited] = useState(initialIsFavorited);
  const [isLoading, setIsLoading] = useState(false);

        const { error } = await supabase
```

```typescript
// src/lib/rate-limit.ts
class RateLimiter {
  private store = new Map<string, { count: number; resetTime: number }>();

  async checkLimit(
    identifier: string,
    maxAttempts: number,
    windowMs: number,
    baseDelayMs: number = 60000 // 1 minute
  ): { allowed: boolean; retryAfter?: number } {
    const now = Date.now();
    const key = `${identifier}`;
    const record = this.store.get(key);

    if (!record || now > record.resetTime) {
      // Reset or create new record
      const retryAfter = baseDelayMs;
      this.store.set(key, {
        count: 1,
        resetTime: now + windowMs,
      });
      return { allowed: true };
    }

    // Check if attempt exceeds limit
    if (record.count >= maxAttempts) {
      const backoffDelay = baseDelayMs * Math.pow(2, record.count - maxAttempts);
      return { 
        allowed: false, 
        retryAfter: Math.max(backoffDelay, 0)
      };
    }

    // Increment attempt
    record.count++;
    return { allowed: true };
  }

  clear(identifier: string): void {
    this.store.delete(identifier);
  }
}

// src/app/api/auth/change-password/route.ts
import { NextRequest, NextResponse } from 'next/server';
import { createClient } from '@/lib/supabase/client';
import { RateLimiter } from '@/lib/rate-limit';

const rateLimiter = new RateLimiter();
const PASSWORD_CHANGE_LIMIT = 1; // 1 attempt per 12 hours
const PASSWORD_CHANGE_WINDOW = 12 * 60 * 60 * 1000; // 12 hours

export async function POST(request: NextRequest) {
  try {
    const { currentPassword, newPassword, userId } = await request.json();

    if (!userId) {
      return NextResponse.json(
        { error: 'Usuário não autenticado' },
        { status: 401 }
      );
    }

    // Check rate limit
    const rateLimit = rateLimiter.checkLimit(
      userId,
      PASSWORD_CHANGE_LIMIT,
      PASSWORD_CHANGE_WINDOW,
      12 * 60 * 60 * 1000 // 12 hours base delay
    );

    if (!rateLimit.allowed) {
      return NextResponse.json(
        { 
          error: 'Muitas tentativas. Tente novamente em algumas horas.',
          retryAfter: rateLimit.retryAfter 
        },
        { status: 429 }
      );
    }

    const supabase = createClient();

    // Verify current password
    const { error: signInError } = await supabase.auth.signInWithPassword({
      email: (await supabase.auth.getUser()).data.user?.email || '',
      password: currentPassword,
    });

    if (signInError) {
      return NextResponse.json(
        { error: 'Senha atual incorreta' },
        { status: 400 }
      );
    }

    // Update password
    const { error: updateError } = await supabase.auth.updateUser({
      password: newPassword,
```

```typescript
  name: string;
  realPrice: string;

}

  id: string;
  marketplace: string;
  url: string;
}

export function ProductGrid({ products }: { products: Product[] }) {
  const [affiliateLinks, setAffiliateLinks] = useState<Record<string, AffiliateLink[]>>({});
```

```typescript
export function Pagination({

  limit,

  currentPage: number;
  totalPages: number;
  limit: number;

    const params = new URLSearchParams();

    params.set('limit', itemsPerPage.toString());
```

```typescript
// Client-side WebP conversion
async function convertToWebP(file: File): Promise<Blob> {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.onload = (e) => {
      const img = new Image();
      img.onload = () => {
        const canvas = document.createElement('canvas');
        canvas.width = img.width;
        canvas.height = img.height;
        const ctx = canvas.getContext('2d');
        if (!ctx) {
          reject(new Error('Failed to create canvas context.'));
          return;
        }
        ctx.drawImage(img, 0, 0);
        canvas.toBlob(
          (blob) => {
            if (!blob) {
              reject(new Error('Failed to convert image to WebP.'));
              return;
            }
            resolve(blob);
          },
          'image/webp',
          0.8
        );
      };
      img.onerror = reject;
      img.src = e.target?.result as string;
    };
    reader.onerror = reject;
    reader.readAsDataURL(file);
  });
}
```

```typescript
export function sanitizeInput(input: string): string {
  // Remove HTML tags and dangerous characters
  return input
    .replace(/<[^>]*>/g, '') // Remove HTML tags
    .replace(/[<>]/g, '') // Remove < and > characters
    .replace(/javascript:/gi, '') // Remove JavaScript protocol
```

### **5. Actions for Comments and Likes**

```typescript
// src/actions/news-engagement.ts
'use server';

import { db } from '@/db';
import { newsComments, newsLikes, newsFeed } from '@/db/schema';
import { createClient } from '@/lib/supabase/server';
import { revalidatePath } from 'next/cache';
import { eq, and } from 'drizzle-orm';
import { isUuid } from '@/lib/uuid';

async function requireAuth() {
  const supabase = await createClient();
  const { data: { user } } = await supabase.auth.getUser();
  if (!user?.id) throw new Error('Acesso negado. Faça login.');
  return user.id;
}

export async function addComment(newsId: string, content: string) {
  const userId = await requireAuth();

  if (!isUuid(newsId)) throw new Error('ID de notícia inválido.');
  if (!content.trim()) throw new Error('Comentário não pode ser vazio.');

  await db.insert(newsComments).values({
    newsId,
    userId,
    content: content.trim(),
  });

  revalidatePath('/noticias');
}

export async function toggleLike(newsId: string) {
  const userId = await requireAuth();

  if (!isUuid(newsId)) throw new Error('ID de notícia inválido.');

  // Check if like already exists
  const [existingLike] = await db
    .select()
    .from(newsLikes)
    .where(and(eq(newsLikes.newsId, newsId), eq(newsLikes.userId, userId)))
    .limit(1);

  if (existingLike) {
    await db.delete(newsLikes).where(eq(newsLikes.id, existingLike.id));
  } else {
    await db.insert(newsLikes).values({ newsId, userId });
  }

  revalidatePath('/noticias');
}

export async function getEngagement(newsId: string) {
  if (!isUuid(newsId)) return { comments: [], likes: 0 };

  const [comments, likes] = await Promise.all([
    db.select().from(newsComments).where(eq(newsComments.newsId, newsId)),
    db.select().from(newsLikes).where(eq(newsLikes.newsId, newsId)),
  ]);

  return { comments, likes: likes.length };
}
```

```typescript
// Client-side WebP conversion
async function convertToWebP(file: File): Promise<Blob> {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.onload = (e) => {
      const img = new Image();
      img.onload = () => {
        const canvas = document.createElement('canvas');
        canvas.width = img.width;
        canvas.height = img.height;
        const ctx = canvas.getContext('2d');
        if (!ctx) {
          reject(new Error('Failed to create canvas context.'));
          return;
        }
        ctx.drawImage(img, 0, 0);
        canvas.toBlob(
          (blob) => {
            if (!blob) {
              reject(new Error('Failed to convert image to WebP.'));
              return;
            }
            resolve(blob);
          },
          'image/webp',
          0.8
        );
      };
      img.onerror = reject;
      img.src = e.target?.result as string;
    };
    reader.onerror = reject;
    reader.readAsDataURL(file);
  });
}
```

```typescript
export function sanitizeInput(input: string): string {
  // Remove HTML tags and dangerous characters
  return input
    .replace(/<[^>]*>/g, '') // Remove HTML tags
    .replace(/[<>]/g, '') // Remove < and > characters
    .replace(/javascript:/gi, '') // Remove JavaScript protocol
    .replace(/on\w+=/gi, '') // Remove event handlers (e.g., onclick=)
```

### **5. Actions for Comments and Likes**
Create a new file `/actions/news-engagement.ts` to handle interactions:

```typescript
// src/actions/news-engagement.ts
'use server';

import { db } from '@/db';
import { newsComments, newsLikes, newsFeed } from '@/db/schema';
import { createClient } from '@/lib/supabase/server';
import { revalidatePath } from 'next/cache';
import { eq, and } from 'drizzle-orm';
import { isUuid } from '@/lib/uuid';

async function requireAuth() {
  const supabase = await createClient();
  const { data: { user } } = await supabase.auth.getUser();
  if (!user?.id) throw new Error('Acesso negado. Faça login.');
  return user.id;
}

export async function addComment(newsId: string, content: string) {
  const userId = await requireAuth();

  if (!isUuid(newsId)) throw new Error('ID de notícia inválido.');
  if (!content.trim()) throw new Error('Comentário não pode ser vazio.');

  await db.insert(newsComments).values({
    newsId,
    userId,
    content: content.trim(),
  });

  revalidatePath('/noticias');
}

export async function toggleLike(newsId: string) {
  const userId = await requireAuth();

  if (!isUuid(newsId)) throw new Error('ID de notícia inválido.');

  // Check if like already exists
  const [existingLike] = await db
    .select()
    .from(newsLikes)
    .where(and(eq(newsLikes.newsId, newsId), eq(newsLikes.userId, userId)))
    .limit(1);

  if (existingLike) {
    await db.delete(newsLikes).where(eq(newsLikes.id, existingLike.id));
  } else {
    await db.insert(newsLikes).values({ newsId, userId });
  }

  revalidatePath('/noticias');
}

export async function getEngagement(newsId: string) {
  if (!isUuid(newsId)) return { comments: [], likes: 0 };

  const [comments, likes] = await Promise.all([
    db.select().from(newsComments).where(eq(newsComments.newsId, newsId)),
    db.select().from(newsLikes).where(eq(newsLikes.newsId, newsId)),
  ]);

  return { comments, likes: likes.length };
}
```

### Fix 1 — Webhook: `x-webhook-signature` + `crypto.createHmac`

```typescript
import { NextRequest, NextResponse } from 'next/server';
import { eq } from 'drizzle-orm';
import { db } from '@/db';
import { subscriptions, userProfiles, businessDirectory } from '@/db/schema';
import crypto from 'crypto';

const WEBHOOK_SECRET = process.env.GATEWAY_WEBHOOK_SECRET;

function extractUserId(payload: Record<string, unknown>): string | null {
  const meta = payload as {
    data?: { metadata?: { user_id?: unknown } };
    external_reference?: unknown;
  };
  const userId =
    meta.data?.metadata?.user_id ??
    meta.external_reference ??
    payload.user_id ??
    null;
  return typeof userId === 'string' && userId.length > 0 ? userId : null;
}

function extractStatus(payload: Record<string, unknown>): string {
  const raw =
    payload.status ??
    (payload as { data?: { status?: unknown } }).data?.status ??
    'active';
  return String(raw).toLowerCase();
}

export async function POST(request: NextRequest) {
  try {
    // 1) Validação da assinatura HMAC via header "x-webhook-signature".
    if (WEBHOOK_SECRET) {
      const signatureHeader = request.headers.get('x-webhook-signature');
      if (!signatureHeader) {
        return NextResponse.json({ error: 'missing signature' }, { status: 401 });
      }

      // Necessário ler o body bruto antes do parse para o HMAC.
      const rawBody = await request.text();
      const expectedHmac = crypto
        .createHmac('sha256', WEBHOOK_SECRET)
        .update(rawBody)
        .digest('hex');

      if (!crypto.timingSafeEqual(Buffer.from(signatureHeader), Buffer.from(expectedHmac))) {
        return NextResponse.json({ error: 'forbidden' }, { status: 401 });
      }

      // Re-parse do JSON a partir do body já lido.
      const payload = JSON.parse(rawBody) as Record<string, unknown>;
      // ... continua o processamento com `payload` ...
    } else {
      const payload = (await request.json()) as Record<string, unknown>;
      // ... processamento sem verificação HMAC (dev-only) ...
    }

    // Reutiliza a lógica de parsing já existente.
    const rawBody = await request.text();
    const payload = JSON.parse(rawBody) as Record<string, unknown>;
    const userId = extractUserId(payload);
    if (!userId) {
      return NextResponse.json({ error: 'missing user reference' }, { status: 422 });
    }

    const gatewayProvider = String(payload.provider ?? 'mercadopago');
    const gatewaySubscriptionId = String(
      (payload as { data?: { id?: unknown } }).data?.id ?? payload.id ?? 'unknown'
    );
    const status = extractStatus(payload);

    const normalizedStatus = ['active', 'past_due', 'canceled'].includes(status)
      ? status
      : 'active';

    const periodEndRaw =
      (payload as { data?: { current_period_end?: number | string } }).data?.current_period_end ??
      payload.current_period_end ??
      null;

    const currentPeriodEnd =
      typeof periodEndRaw === 'number'
        ? new Date(periodEndRaw * 1000)
        : typeof periodEndRaw === 'string'
          ? new Date(periodEndRaw)
          : null;

    await db
      .insert(subscriptions)
      .values({
        userId,
        gatewayProvider,
        gatewaySubscriptionId,
        status: normalizedStatus,
        currentPeriodEnd,
        updatedAt: new Date(),
      })
      .onConflictDoUpdate({
        target: subscriptions.gatewaySubscriptionId,
        set: {
          status: normalizedStatus,
          currentPeriodEnd,
          updatedAt: new Date(),
        },
      });

    const planType = normalizedStatus === 'active' ? 'premium' : 'free';

    await db.update(userProfiles).set({ planType }).where(eq(userProfiles.id, userId));
    await db.update(businessDirectory).set({ planType }).where(eq(businessDirectory.userId, userId));

    return NextResponse.json({ received: true }, { status: 200 });
  } catch (error) {
    console.error('Webhook error:', error);
    return NextResponse.json({ error: 'invalid payload' }, { status: 400 });
  }
}
```

**justificativa:** `crypto.createHmac('sha256', ...)` gera o digest esperado; `crypto.timingSafeEqual` compara em tempo constante, eliminando ataque de timing. O header agora é `x-webhook-signature` conforme exigido.

### Fix 2 — LGPD: transação atômica abrangendo `luthier_leads`, `luthier_reviews`, `subscriptions`

#### `deleteUserAccount` corrigido

```typescript
'use server';

import { eq } from 'drizzle-orm';
import { redirect } from 'next/navigation';
import { revalidatePath } from 'next/cache';
import { createClient } from '@/lib/supabase/server';
import { db } from '@/db';
import {

} from '@/db/schema';

export async function exportUserData(): Promise<string> {
  const supabase = await createClient();
  const { data: { user } } = await supabase.auth.getUser();
  if (!user?.id) {
    throw new Error('Sessão inválida. Faça login novamente.');
  }

  const [profileRows, favStrings, favLuthiers, sReviews, lReviews, notifs, leads, subs] = await Promise.all([
    db.select().from(userProfiles).where(eq(userProfiles.id, user.id)),

    db.select().from(favoriteLuthiers).where(eq(favoriteLuthiers.userId, user.id)),
    db.select().from(stringReviews).where(eq(stringReviews.userId, user.id)),
    db.select().from(luthierReviews).where(eq(luthierReviews.userId, user.id)),
    db.select().from(notifications).where(eq(notifications.userId, user.id)),
    db.select().from(luthierLeads).where(eq(luthierLeads.userId, user.id)),
    db.select().from(subscriptions).where(eq(subscriptions.userId, user.id)),
  ]);

  return JSON.stringify(
    {
      exportedAt: new Date().toISOString(),
      profile: profileRows[0] ?? null,
      favoriteStrings: favStrings,
      favoriteLuthiers: favLuthiers,
      stringReviews: sReviews,
      luthierReviews: lReviews,
      notifications: notifs,
      luthierLeads: leads,
      subscriptions: subs,
    },
    null,
    2,
  );
}

export async function deleteUserAccount(): Promise<void> {
  const supabase = await createClient();
  const { data: { user } } = await supabase.auth.getUser();
  if (!user?.id) {
    redirect('/login');
  }

  // Transação atômica: toda a deleção LGPD reverte se qualquer passo falhar.
  await db.transaction(async (tx) => {
    // 1. Anonimização do perfil.
    await tx.update(userProfiles).set({
      email: `anon_${user.id}@anonimizado.local`,
      fullName: 'Usuário Anonimizado',
      businessName: null,
      city: null,
      state: null,
      bio: null,
    }).where(eq(userProfiles.id, user.id));

    // 2. Anonimização da vitrine pública.
    await tx.update(businessDirectory).set({
      businessName: 'Estabelecimento Anonimizado',
      city: 'Anonimizado',
      state: 'NA',
      whatsapp: null,
    }).where(eq(businessDirectory.userId, user.id));

    // 3. Exclusão de dados sem valor histórico preservável.
    await tx.delete(favoriteStrings).where(eq(favoriteStrings.userId, user.id));
    await tx.delete(favoriteLuthiers).where(eq(favoriteLuthiers.userId, user.id));
    await tx.delete(notifications).where(eq(notifications.userId, user.id));

    // 4. Exclusão explícita das tabelas exigidas pela LGPD.
    await tx.delete(luthierLeads).where(eq(luthierLeads.userId, user.id));
    await tx.delete(luthierReviews).where(eq(luthierReviews.userId, user.id));
    await tx.delete(stringReviews).where(eq(stringReviews.userId, user.id));
    await tx.delete(subscriptions).where(eq(subscriptions.userId, user.id));
  });

  await supabase.auth.signOut();
  revalidatePath('/');
  redirect('/login');
}
```

**justificativa:** `db.transaction(...)` garante atomicidade — se qualquer `DELETE` ou `UPDATE` falhar, o Drizzle reverte toda a transação via rollback do Postgres, evitando estado parcial de anonimização. As tabelas `luthier_leads`, `luthier_reviews` e `subscriptions` agora são explicitamente incluídas, junto com `stringReviews` e `notifications` que também pertencem ao escopo LGPD do usuário.
